package com.covernotifier.flip

import android.content.BroadcastReceiver
import android.content.Context
import android.content.Intent
import android.content.IntentFilter
import android.hardware.display.DisplayManager
import android.util.Log
import android.view.Display

/**
 * Tracks whether the flip is closed ("covered").
 *
 * Ground truth is the framework's sticky broadcast android.os.action.HALL_COVER_CHANGED with a
 * boolean "cover" extra. We deliberately avoid PowerManager.getIsHallCover(), which is a vendor
 * hidden API and would need reflection past Android 11's blocklist.
 *
 * Fallback when no sticky value is cached yet: display 1 is only powered while covered, so its
 * Display.state is a good proxy.
 */
class FlipMonitor(
    private val ctx: Context,
    private val onChange: (covered: Boolean) -> Unit
) {
    companion object {
        const val ACTION_HALL = "android.os.action.HALL_COVER_CHANGED"
        const val EXTRA_COVER = "cover"
        private const val TAG = "CoverNotifier/Flip"
        const val COVER_DISPLAY_ID = 1
    }

    @Volatile
    var isCovered: Boolean = false
        private set

    private var registered = false

    private val receiver = object : BroadcastReceiver() {
        override fun onReceive(c: Context?, intent: Intent?) {
            if (intent?.action != ACTION_HALL) return
            val covered = intent.getBooleanExtra(EXTRA_COVER, false)
            if (covered == isCovered) return
            isCovered = covered
            Log.i(TAG, "cover=" + covered)
            onChange(covered)
        }
    }

    fun start() {
        if (registered) return
        val filter = IntentFilter(ACTION_HALL)
        // Peek at the sticky value before subscribing so we start with real state, not a guess.
        val sticky = try {
            ctx.registerReceiver(null, filter)
        } catch (t: Throwable) {
            null
        }
        isCovered = when {
            sticky != null && sticky.hasExtra(EXTRA_COVER) ->
                sticky.getBooleanExtra(EXTRA_COVER, false)
            else -> probeByDisplay()
        }
        ctx.registerReceiver(receiver, filter)
        registered = true
        Log.i(TAG, "start, initial cover=" + isCovered)
        onChange(isCovered)
    }

    fun stop() {
        if (!registered) return
        try {
            ctx.unregisterReceiver(receiver)
        } catch (_: Throwable) {
        }
        registered = false
    }

    /** True when the external display is powered, which on this device means the flip is shut. */
    fun probeByDisplay(): Boolean {
        val dm = ctx.getSystemService(DisplayManager::class.java) ?: return false
        val d = dm.getDisplay(COVER_DISPLAY_ID) ?: return false
        return d.state == Display.STATE_ON
    }
}
